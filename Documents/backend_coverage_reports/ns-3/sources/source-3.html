         String cardValue = faceCardConverter(getValue());

     public String suitConverter(char cardSuit) {
         if (cardSuit == 'h') {
             return "hearts";
         } else if (cardSuit == 'd') {
             return "diamonds";
         } else if (cardSuit == 's') {
             return "spades";
         } else if (cardSuit == 'c') {
             return "clubs";
         } else {
             return "UNRECOGNIZED SUIT";
         }
     }

             return returnValue;
